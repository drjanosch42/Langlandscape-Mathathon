Let $G,\Gamma,\iota,U,R,V,\rho$ be as in \cref{X.hmod.def:equivariant-functions} with $\rho(1)=1$, and let $R_0$ be a commutative ring such
that $V$ is an $R_0$-module and every $\rho(\gamma)$ is $R_0$-linear. Let $\mathcal F=\mathcal F(\iota,U,\rho)$. Then for
$\varphi,\varphi_1,\varphi_2\in\mathcal H(G,U;R_0)$, $r\in R_0$ and $f,g\in\mathcal F$:
\begin{enumerate}[label=(\alph*)]
\item $\varphi\cdot f\in\mathcal F$;
\item $\mathbf 1_U\cdot f=f$;
\item $(\varphi_1*\varphi_2)\cdot f=\varphi_1\cdot(\varphi_2\cdot f)$;
\item $(\varphi_1+\varphi_2)\cdot f=\varphi_1\cdot f+\varphi_2\cdot f$ and $(r\varphi)\cdot f=r\,(\varphi\cdot f)$;
\item $\varphi\cdot(f+g)=\varphi\cdot f+\varphi\cdot g$.
\end{enumerate}

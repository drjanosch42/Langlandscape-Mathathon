Let $F$ be a non-archimedean local field, $G=\GL_2(F)$, $K=K_F$, and let $\chi_1,\chi_2$ be quasi-characters of
$F^\times$.
\begin{enumerate}[label=(\alph*)]
\item If $\chi_1$ and $\chi_2$ are unramified, then $f^\circ_\chi\in I(\chi_1,\chi_2)^K$, $f^\circ_\chi(1)=1$, and
  $I(\chi_1,\chi_2)^K=\C f^\circ_\chi$; in particular $\dim I(\chi_1,\chi_2)^K=1$.
\item If $\chi_1$ or $\chi_2$ is ramified (not unramified), then $I(\chi_1,\chi_2)^K=0$.
\end{enumerate}

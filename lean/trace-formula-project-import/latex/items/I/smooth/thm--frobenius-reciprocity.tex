Let $F$ be a non-archimedean local field, $G=\GL_2(F)$, let $\chi_1,\chi_2$ be quasi-characters of $F^\times$, and let
$\psi$ be the character $\psi(\diag(t_1,t_2))=\chi_1(t_1)\chi_2(t_2)\abs{t_1/t_2}_F^{1/2}$ of $T(F)$. For a smooth
representation $(\pi,V)$ of $G$ write $\Hom_{T(F)}(V_N,\psi)$ for the space of $\C$-linear maps $\lambda\colon V_N\to\C$
with $\lambda(\pi_N(t)\xi)=\psi(t)\lambda(\xi)$ for $t\in T(F)$, $\xi\in V_N$, and $\Hom_G(V,I(\chi_1,\chi_2))$ for the
space of $G$-equivariant $\C$-linear maps $V\to I(\chi_1,\chi_2)$ (with $\pi_\chi$ on the target).
\begin{enumerate}[label=(\alph*)]
\item For $\Phi\in\Hom_G(V,I(\chi_1,\chi_2))$ the map
  \[
    \Phi^\flat\colon V_N\to\C,\qquad\Phi^\flat(v+V(N))=\Phi(v)(1),
  \]
  is well defined and lies in $\Hom_{T(F)}(V_N,\psi)$.
\item The map $\Phi\mapsto\Phi^\flat$ is a $\C$-linear bijection
  $\Hom_G(V,I(\chi_1,\chi_2))\to\Hom_{T(F)}(V_N,\psi)$.
\item It is natural in $V$: for every $G$-equivariant linear map $u\colon V'\to V$ from a smooth representation $V'$,
  $(\Phi\circ u)^\flat=\Phi^\flat\circ u_N$.
\end{enumerate}

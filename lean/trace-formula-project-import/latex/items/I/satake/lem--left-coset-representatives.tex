Let $F$ be a non-archimedean local field with uniformizer $\unif$, $G=\GL_2(F)$ and $K=K_F$. For $x\in F^\times$ let
$\operatorname{ord}(x)\in\Z$ be the integer with $x\in\unif^{\operatorname{ord}(x)}\Ocal_F^\times$, and put
$\operatorname{ord}(0)=+\infty$. For $i,j\in\Z$ and $x\in F$ put
$h_{i,j}(x)=\begin{psmallmatrix}\unif^i&x\\0&\unif^j\end{psmallmatrix}\in G$, and for every $i\in\Z$ fix a set
$\mathcal R_i\subseteq F$ of representatives of the cosets of $\unif^i\Ocal_F$ in $F$.
\begin{enumerate}[label=(\alph*)]
\item Every left coset $gK$ ($g\in G$) contains exactly one matrix $h_{i,j}(x)$ with $i,j\in\Z$ and $x\in\mathcal R_i$.
\item Let $a\ge b$ be integers, $i,j\in\Z$ and $x\in F$. Then $h_{i,j}(x)\in K\,t_{a,b}\,K$ if and only if $i+j=a+b$
  and $\min(i,j,\operatorname{ord}(x))=b$.
\end{enumerate}

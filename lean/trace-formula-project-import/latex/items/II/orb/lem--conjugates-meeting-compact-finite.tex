Let $\gamma\in\GL_2(\Q)$ and let $C\subseteq G(\A)$ be compact. Then:
\begin{enumerate}
  \item[(a)] the set $\{\delta^{-1}\gamma\delta:\delta\in G(\Q)\}\cap C$ is finite;
  \item[(b)] there are finitely many $\delta_1,\dots,\delta_n\in G(\Q)$ such that
  $\{\delta\in G(\Q):\delta^{-1}\gamma\delta\in C\}=\bigcup_{i=1}^nG_\gamma(\Q)\,\delta_i$.
\end{enumerate}

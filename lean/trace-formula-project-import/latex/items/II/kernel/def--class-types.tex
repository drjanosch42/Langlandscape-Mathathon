A geometric class $\mathcal O$ with polynomial $P=X^2-tX+d$ (\cref{II.kernel.def:geometric-class}) is
\begin{enumerate}
  \item[(i)] \emph{central-unipotent} if $P=(X-z)^2$ for some $z\in\Q^\times$; we then write $\mathcal O=\mathcal O_z$;
  \item[(ii)] \emph{hyperbolic} if $P=(X-a)(X-b)$ for some $a,b\in\Q^\times$ with $a\neq b$; we then write $\mathcal O=\mathcal O_{a,b}$ ($=\mathcal O_{b,a}$);
  \item[(iii)] \emph{elliptic} if $P$ is irreducible in $\Q[X]$.
\end{enumerate}

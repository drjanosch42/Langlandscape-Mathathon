Let $a,b\in\Z$, let $R$ be a commutative ring and $x,y\in\Bq_R$, with $R$ identified with $R\cdot1\subseteq\Bq_R$. Then:
\begin{enumerate}[label=(\alph*)]
\item $x\bar x=\bar x x=\nrd(x)$ and $x+\bar x=\trd(x)$;
\item $x^2-\trd(x)\,x+\nrd(x)=0$;
\item $\nrd(xy)=\nrd(x)\nrd(y)$ and $\trd(xy)=\trd(yx)$;
\item for a ring homomorphism $f\colon R\to S$ with base change $f_{\Bq}\colon\Bq_R\to\Bq_S$ (\cref{III.qalg.def:quaternion-algebra}):
  $\nrd(f_{\Bq}(x))=f(\nrd(x))$ and $\trd(f_{\Bq}(x))=f(\trd(x))$.
\end{enumerate}

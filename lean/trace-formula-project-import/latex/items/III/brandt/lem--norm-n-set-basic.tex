Let $a,b\in\Z$, $\Oord$ an order of $\Bq$ and $n\ge1$. Then:
\begin{enumerate}[label=(\alph*)]
\item $uyv\in\Ohat_n$ for $y\in\Ohat_n$ and $u,v\in\Ohat^\times$;
\item $\bar y\in\Ohat_n$ for $y\in\Ohat_n$;
\item $\Ohat_1=\Ohat^\times$;
\item $\Ohat_n$ meets only finitely many cosets $y\Ohat^\times$;
\item hence the indicator $\mathbf 1_{\Ohat_n}$ lies in $\mathcal H(\Bhat^\times,\Ohat^\times;R_0)$ for every commutative ring $R_0$.
\end{enumerate}

Let $a,b\in\Z$, let $\Oord$ be an order of $\Bq$, and for $x\in\Bhat^\times$ let $L_x=\iota^{-1}(x\Ohat)$. Then:
\begin{enumerate}[label=(\alph*)]
\item $L_x\Oord\subseteq L_x$ ($L_x$ is a right $\Oord$-module);
\item $L_{\iota(\gamma)xu}=\gamma L_x$ for $\gamma\in\Bq^\times$ and $u\in\Ohat^\times$;
\item if $x,y\in\Bhat^\times$ and $L_x=\gamma L_y$ for some $\gamma\in\Bq^\times$, then $[x]=[y]$ in $X(\Oord)$;
\item let $L\subseteq\Bq$ be a lattice (a finitely generated $\Z$-submodule with $\Q L=\Bq$) with $L\Oord\subseteq L$, $S$ a finite set of primes and $\beta_p\in\Bq_{\Qp}^\times$
  for $p\in S$, such that $L_p=\beta_p\Oord_p$ for $p\in S$ and $L_p=\Oord_p$ for every prime $p\notin S$ (\cref{III.qalg.def:local-order}). Then
  $L=L_x$ for the element $x\in\Bhat^\times$ whose $p$-component is $\beta_p$ for $p\in S$ and $1$ for $p\notin S$.
\end{enumerate}

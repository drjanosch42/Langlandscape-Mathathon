Let $D\in\N$ be squarefree. For a positive integer $m$ let $\omega(m)$ be the number of distinct primes dividing $m$, and for a positive divisor
$d$ of $D$ put $s_d=(-2)^{\omega(D/d)}\in\Z$. Then, with $d$ running over the positive divisors of $D$:
\begin{enumerate}[label=(\alph*)]
\item $\sum_{d\mid D}s_d=(-1)^{\omega(D)}$;
\item if $D>1$, then $\sum_{d\mid D}s_d\,2^{\omega(d)}=0$;
\item if $M\in\N$ and $\gcd(D,M)=1$, then $\sum_{d\mid D}s_d\,\dedpsi(dM)=\dedpsi(M)\prod_{p\mid D}(p-1)$ in $\Q$, with $\dedpsi$ the Dedekind
  function (\cref{III.ef.def:dedekind-psi}) and the product over the primes dividing $D$.
\end{enumerate}

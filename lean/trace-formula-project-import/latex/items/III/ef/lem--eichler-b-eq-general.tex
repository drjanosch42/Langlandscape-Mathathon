For all $D,M,k,n\in\N$ and every function $F\colon\Z\to\Q$,
\[
  \EichB(D,M,k,n;F)=\EichB_{\mathrm{gen}}\bigl(D,k,n;\dedpsi(M),F'\bigr),\qquad F'(t,f,\Delta)=F(\Delta),
\]
with $\EichB$, $\EichB_{\mathrm{gen}}$ and $\dedpsi$ as in
\cref{III.ef.def:eichler-b-expression,III.ef.def:eichler-b-expression-general,III.ef.def:dedekind-psi}.

Let $K$ be a field of characteristic $0$ and $m\in\N$. Then:
\begin{enumerate}[label=(\alph*)]
\item $\beta_m(\rho_m(A)v,w)=\beta_m(v,\rho_m(\operatorname{adj}A)w)$ for all $A\in\Mat_2(K)$ and $v,w\in V_m(K)$, where
  $\operatorname{adj}\begin{psmallmatrix}A_{11}&A_{12}\\A_{21}&A_{22}\end{psmallmatrix}=\begin{psmallmatrix}A_{22}&-A_{12}\\-A_{21}&A_{11}\end{psmallmatrix}$;
\item $\beta_m$ is nondegenerate: if $\beta_m(v,w)=0$ for all $w$, then $v=0$;
\item if $m$ is even, $\beta_m$ is symmetric;
\item $\beta_m(\rho_m(A)v,\rho_m(A)w)=\det(A)^m\,\beta_m(v,w)$ for all $A\in\Mat_2(K)$ and $v,w\in V_m(K)$.
\end{enumerate}

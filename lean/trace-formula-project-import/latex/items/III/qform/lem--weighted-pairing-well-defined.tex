Let $a,b\in\Z$ with $a<0$ and $b<0$, $\Oord$ an order of $\Bq$, $K$ a field of characteristic $0$, $s$ a splitting of $\Bq$ over $K$ and
$k\in\N$. On $M_k(\Oord;s)$:
\begin{enumerate}[label=(\alph*)]
\item for any elements $y_c\in\Bhat^\times$ with $[y_c]=c$, $\langle f,g\rangle=\sum_c\abs{\Gamma_{y_c}}^{-1}\nu(y_c)^{k-2}\beta_{k-2}(f(y_c),g(y_c))$;
\item if $k$ is even, $\langle f,g\rangle=\langle g,f\rangle$;
\item if $\langle f,g\rangle=0$ for all $g\in M_k(\Oord;s)$, then $f=0$.
\end{enumerate}

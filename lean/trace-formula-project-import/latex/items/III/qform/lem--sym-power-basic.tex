Let $K$ be a field and $m\in\N$. Then:
\begin{enumerate}[label=(\alph*)]
\item the monomials $u_t=x^{m-t}y^t$, $0\le t\le m$, form a basis of $V_m(K)$; in particular $\dim_KV_m(K)=m+1$;
\item $\rho_m(AB)=\rho_m(A)\rho_m(B)$ for $A,B\in\Mat_2(K)$, and $\rho_m(1)=1$;
\item $\rho_m(c\cdot1)=c^m$ for $c\in K$;
\item if $m$ is even, then $\rho_m(-1)=1$.
\end{enumerate}

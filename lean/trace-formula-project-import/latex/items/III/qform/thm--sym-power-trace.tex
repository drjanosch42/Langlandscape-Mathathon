Let $K$ be a field. Then
\begin{enumerate}[label=(\alph*)]
\item $\tr\rho_m(A)=P_{m+2}(\tr A,\det A)$ for every $m\in\N$ and every $A\in\Mat_2(K)$, with $P_k$ over $K$ as in
  \cref{III.qform.def:hecke-polynomial-ring};
\item for $k\in\N$, $t\in\Z$ and $n\in\N$, the image in $K$ of the integer $P_k(t,n)$ of \cref{III.ef.def:hecke-polynomial} is $P_k(t,n)$ computed in $K$.
\end{enumerate}

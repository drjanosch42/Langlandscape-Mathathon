Let $a,b\in\Z$ and let $\Oord$ be an order of $\Bq$. Then:
\begin{enumerate}[label=(\alph*)]
\item for every $\Z$-basis $e_1,\dots,e_4$ of $\Oord$ (a finite index type), $d(\Oord)=\det\bigl(\trd(e_\mu\bar e_\nu)\bigr)_{\mu,\nu}$, so $d(\Oord)$ does not depend on the basis
  (\cref{III.qord.def:discriminant});
\item $d(\Oord)\in\Z$;
\item if $a<0$ and $b<0$, then $d(\Oord)>0$.
\end{enumerate}

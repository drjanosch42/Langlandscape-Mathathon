Let $a,b\in\Z$, let $L\subseteq\Bq$ be a lattice (a finitely generated $\Z$-submodule with $\Q L=\Bq$) and $p$ a prime; give $\Bq_{\Qp}$ the topology for which the
coordinate map $\Bq_{\Qp}\to\Qp^4$ is a homeomorphism. Then:
\begin{enumerate}[label=(\alph*)]
\item $L_p$ is a compact open $\Zp$-submodule of $\Bq_{\Qp}$ that spans $\Bq_{\Qp}$ over $\Qp$;
\item for every $\Z$-basis $e_1,\dots,e_4$ of $L$, $L_p$ is the $\Zp$-span of the images of $e_1,\dots,e_4$;
\item if $L'\subseteq L$ is a lattice and $p\nmid[L:L']$, then $L'_p=L_p$;
\item if $L=\Oord$ is an order of $\Bq$ (\cref{III.qord.def:order}), then $\Oord_p$ is a subring of $\Bq_{\Qp}$; with (a), $\Oord_p$ is a $\Zp$-order.
\end{enumerate}

In the situation of \cref{III.brandt.def:local-hecke-embedding} assume moreover $p\nmid M$, and let $K_p=\GL_2(\Zp)$. Then:
\begin{enumerate}[label=(\alph*)]
\item $\Theta_p(\Phi)\in\mathcal H(\Bhat^\times,\Ohat^\times;\C)$ for every $\Phi\in\mathcal H(\GL_2(\Qp),K_p)$;
\item $\Theta_p(\mathbf 1_{K_p})=\mathbf 1_{\Ohat^\times}$ and $\Theta_p(\Phi_1*\Phi_2)=\Theta_p(\Phi_1)*\Theta_p(\Phi_2)$;
\item $\Theta_p\bigl(\sum_{a+b=r,\ a\ge b\ge0}c_{a,b}\bigr)=\mathbf 1_{\Ohat_{p^r}}$ for every $r\ge0$ (\cref{I.satake.def:hecke-basis});
\item $\Theta_p(c_{1,1})=\mathbf 1_{\iota(p)\Ohat^\times}$.
\end{enumerate}

Let $a,b\in\Z$; lattices are finitely generated $\Z$-submodules $L\subseteq\Bq$ with $\Q L=\Bq$.
\begin{enumerate}[label=(\alph*)]
\item For every lattice $L$, $\iota^{-1}(\Lhat)=L$.
\item For every order $\Oord$ of $\Bq$ and $x\in\Bhat^\times$, $L_x=\iota^{-1}(x\Ohat)$ is a lattice, and $\widehat{L_x}=x\Ohat$.
\item For lattices $L,L'$: if $L_p=L'_p$ for every prime $p$ (\cref{III.qalg.def:local-order}), then $L=L'$; and $\Lhat=\widehat{L'}$ if and only if
  $L=L'$.
\end{enumerate}

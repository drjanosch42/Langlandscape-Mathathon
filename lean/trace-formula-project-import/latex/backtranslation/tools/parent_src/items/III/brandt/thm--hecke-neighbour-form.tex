Let $a,b\in\Z$, $\Oord$ an order of $\Bq$, $K$ a field of characteristic $0$, $s$ a splitting of $\Bq$ over $K$, $k\in\N$, $n\ge1$ and
$x\in\Bhat^\times$. Then:
\begin{enumerate}[label=(\alph*)]
\item for $y\in\Ohat_n$: $L_{xy}\subseteq L_x$ and $[L_x:L_{xy}]=n^2$;
\item for $y,y'\in\Ohat_n$: $L_{xy}=L_{xy'}$ if and only if $y\Ohat^\times=y'\Ohat^\times$;
\item if $J\subseteq L_x$ is a $\Z$-submodule with $J\Oord\subseteq J$ and $[L_x:J]=n^2$ such that for every prime $p$ there is
  $\beta_p\in\Bq_{\Qp}^\times$ with $J_p=\beta_p\Oord_p$, then $J=L_{xy}$ for some $y\in\Ohat_n$;
\item if $z,x'\in\Bhat^\times$ and $\gamma\in\Bq^\times$ with $L_z=\gamma L_{x'}$, then $f(z)=\rho_k^s(\gamma)f(x')$ for every $f\in M_k(\Oord;s)$.
\end{enumerate}

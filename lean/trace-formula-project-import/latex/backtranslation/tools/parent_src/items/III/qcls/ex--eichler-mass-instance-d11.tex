Let $(a,b)=(-1,-11)$ and let $\Oord_{11}=\Z\,\tfrac{1+j}2+\Z\,\tfrac{i+k}2+\Z j+\Z k\subseteq\Bq$. Then:
\begin{enumerate}[label=(\alph*)]
\item $\operatorname{RamEx}(-1,-11,11)$ (\cref{III.qalg.def:ramified-exactly});
\item $\Oord_{11}$ is a maximal order of $\Bq$, and $d(\Oord_{11})=121$;
\item $\abs{X(\Oord_{11})}=2$; the class $[1]$ has $\abs{\Gamma_1}=\abs{\Oord_{11}^\times}=4$, and the other class $[x]$ has $\abs{\Gamma_x}=6$;
\item $\Mass(\Oord_{11})=\frac14+\frac16=\frac5{12}$;
\item the hypothesis H-EM$(-1,-11,11,\Oord_{11})$ of \cref{III.qcls.hyp:eichler-mass} holds.
\end{enumerate}

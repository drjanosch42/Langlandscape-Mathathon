For all $N,k,n\in\N$ and every function $F\colon\Z\to\Q$,
\[
  \LamES(N,k,n;F)=\LamES_{\mathrm{gen}}\bigl(k,n;\dedpsi(N),F',\beta_N\bigr),\qquad F'(t,f,\Delta)=F(\Delta),\quad\beta_N(a,a')=\cbd_N(a-a'),
\]
with $\LamES$, $\LamES_{\mathrm{gen}}$, $\dedpsi$ and $\cbd$ as in
\cref{III.ef.def:lefschetz-es-expression,III.ef.def:lefschetz-es-expression-general,III.ef.def:dedekind-psi,III.ef.def:boundary-divisor-sum}.

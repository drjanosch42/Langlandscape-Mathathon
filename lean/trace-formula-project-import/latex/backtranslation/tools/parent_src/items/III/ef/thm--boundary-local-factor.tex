Let $\cbd$ be as in \cref{III.ef.def:boundary-divisor-sum} and $m\in\Z$.
\begin{enumerate}[label=(\alph*)]
\item For every prime $p$ and every $e\in\N$,
  \[
    \cbd_{p^e}(m)=\sum_{j=0}^{e}\bigl[p^{\min(j,e-j)}\mid m\bigr]\,\varphi\bigl(p^{\min(j,e-j)}\bigr),
  \]
  where $[P]$ is $1$ if the statement $P$ holds and $0$ otherwise, and $\varphi$ is Euler's function.
\item $\cbd_p(m)=2$ for every prime $p$.
\item $\cbd_N(m)=2^{\omega(N)}$ for every squarefree $N\in\N$, where $\omega(N)$ is the number of distinct primes dividing $N$.
\end{enumerate}

For $N,k,n\in\N$ put
\[
  A_2(N,k,n)=-\frac12\sum_{\substack{t\in\Z\\ t^2<4n}}P_k(t,n)\sum_{f\in\Ford(t,n)}\hw(\Delta_f)\,\embmu_N(t,f,n)\in\Q,\qquad \Delta_f=\frac{t^2-4n}{f^2},
\]
with $P_k$, $\Ford$, $\hw$ and $\embmu$ as in
\cref{III.ef.def:hecke-polynomial,III.ef.def:order-index-set,III.ef.def:weighted-class-number,III.ef.def:cohen-mu}.

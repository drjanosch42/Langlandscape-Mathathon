Let $a,b\in\Z$ with $a<0$ and $b<0$, $\Oord$ an order of $\Bq$, $K$ a field of characteristic $0$ and $s$ a splitting of $\Bq$ over $K$.
\begin{enumerate}[label=(\alph*)]
\item $M_k(\Oord;s)$ is finite-dimensional for every $k$.
\item Let $k\ge2$ and let $\Gamma\le\Bq^\times$ be a finite subgroup with $\nrd(\gamma)=1$ for all $\gamma\in\Gamma$. Then
  \[\abs{\Gamma}\cdot\dim_KV_{k-2}(K)^{\Gamma}=\sum_{\gamma\in\Gamma}P_k(\trd\gamma,1)\qquad\text{in }K.\]
\item $\dim_KM_2(\Oord;s)=\abs{X(\Oord)}$.
\end{enumerate}

Let $a,b,D,M,\Oord_0,\phi$ and $\Oord=\Oord_0(M;\phi)$ be as in \cref{III.brandt.thm:hecke-coset-count}, $K$ a field of characteristic $0$, $s$ a
splitting of $\Bq$ over $K$, and $n\ge1$ with $\gcd(n,DM)=1$. Then, in weight $2$:
\begin{enumerate}[label=(\alph*)]
\item $T_n\mathbf 1=\sigma_1(n)\,\mathbf 1$;
\item if $f\in M_2(\Oord;s)$ and $\langle f,\mathbf 1\rangle=0$, then $\langle T_nf,\mathbf 1\rangle=0$;
\item for every $f\in M_2(\Oord;s)$ there is exactly one $c\in K$ with $\langle f-c\mathbf 1,\mathbf 1\rangle=0$.
\end{enumerate}

Let $a,b\in\Z$ with $a<0$ and $b<0$, $\Oord$ an order of $\Bq$, $K$ a field of characteristic $0$, $s$ a splitting of $\Bq$ over $K$ and
$n\ge1$; let $x_c$ be the chosen representatives. Then:
\begin{enumerate}[label=(\alph*)]
\item $(T_nf)(x_c)=\sum_{c'\in X(\Oord)}\Brandt n_{cc'}\,f(x_{c'})$ for every $f\in M_2(\Oord;s)$ and $c\in X(\Oord)$;
\item $\abs{\Gamma_{x_{c'}}}\cdot\Brandt n_{cc'}=\#\bigl\{\gamma\in\Bq^\times:\gamma L_{x_{c'}}\subseteq L_{x_c},\ \nrd\gamma=n\,\nu(x_{c'})/\nu(x_c)\bigr\}$
  for all $c,c'\in X(\Oord)$;
\item for every $x\in\Bhat^\times$ with $[x]=c$: $\Brandt n_{cc'}=\#\{y\Ohat^\times\subseteq\Ohat_n:[xy]=c'\}$.
\end{enumerate}

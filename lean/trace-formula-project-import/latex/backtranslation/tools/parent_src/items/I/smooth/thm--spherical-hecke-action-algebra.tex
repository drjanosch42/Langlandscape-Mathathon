Let $F$ be a non-archimedean local field, $G=\GL_2(F)$, $K=K_F$, and let $\pi$ be a representation of $G$ on a complex
vector space $V$.
\begin{enumerate}[label=(\alph*)]
\item $\pi(f_1*f_2)=\pi(f_1)\circ\pi(f_2)$ on $V^K$ for all $f_1,f_2\in\mathcal H(G,K)$, and $\pi(\mathbf 1_K)$ is the
  identity of $V^K$. Thus $V^K$ is a module over $\mathcal H(G,K)$.
\item Suppose $\pi$ is smooth, and let $\mu$ be a left Haar measure on $G$ (with its Borel $\sigma$-algebra). For
  $f\in\mathcal H(G,K)$ and $v\in V^K$, the vector $\int_Gf(g)\pi(g)v\,\dd\mu(g)$ of \cref{I.rep.def:smooth-hecke-action}
  equals $\mu(K)\,\pi(f)v$.
\end{enumerate}

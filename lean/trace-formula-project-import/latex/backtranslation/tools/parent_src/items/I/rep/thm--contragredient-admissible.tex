Let $G$ be a locally profinite group and $\pi$ an admissible representation of $G$ on $V$, with contragredient
$(\tilde\pi,\tilde V)$.
\begin{enumerate}[label=(\alph*)]
\item For every compact open subgroup $K'\subseteq G$, restriction $\lambda\mapsto\lambda|_{V^{K'}}$ is a linear
  isomorphism $\tilde V^{K'}\to(V^{K'})^*$.
\item $\tilde\pi$ is admissible.
\item The map $V\to\tilde{\tilde V}$, $v\mapsto(\lambda\mapsto\lambda(v))$, is well defined, $\C$-linear, bijective and
  $G$-equivariant.
\end{enumerate}

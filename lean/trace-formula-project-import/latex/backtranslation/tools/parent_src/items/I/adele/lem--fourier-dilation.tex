Let $\lambda$ be a Haar measure on $\A$ and $a\in\A^\times$. For $\Phi\colon\A\to\C$ put $\Phi_a(x)=\Phi(ax)$. Then:
\begin{enumerate}
\item $\lambda(aE)=\abs{a}_\A\,\lambda(E)$ for every Borel set $E\subseteq\A$;
\item for every $\Phi\colon\A\to\C$, the Fourier transform of $\Phi_a$ with respect to $\lambda$ is
  \[
    \widehat{\Phi_a}(y)=\abs{a}_\A^{-1}\,\hat\Phi(a^{-1}y)\qquad(y\in\A),
  \]
  where $\hat\Phi$ is the Fourier transform of $\Phi$ with respect to $\lambda$;
\item if $\Phi\in\Schw(\A)$, then $\Phi_a\in\Schw(\A)$.
\end{enumerate}

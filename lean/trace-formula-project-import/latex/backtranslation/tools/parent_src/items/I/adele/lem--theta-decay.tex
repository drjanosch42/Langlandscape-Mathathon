Let $\Phi\in\Schw(\A)$. Then:
\begin{enumerate}
\item for all $r>0$ and $u\in\Zhat^\times$ the series $\sum_{q\in\Q}\Phi(q\cdot(r,u))$ converges absolutely;
\item there is a constant $C$ such that $\sum_{q\in\Q^\times}\abs{\Phi(q\cdot(r,u))}\le C\,r^{-1}$ for all
  $r\in(0,1]$ and $u\in\Zhat^\times$, and for every $N\ge0$ a constant $C_N$ such that
  $\sum_{q\in\Q^\times}\abs{\Phi(q\cdot(r,u))}\le C_N\,r^{-N}$ for all $r\ge1$ and $u\in\Zhat^\times$;
\item for every continuous homomorphism $\omega\colon\Zhat^\times\to\C^\times$, the function $\theta_{\Phi,\omega}$ is
  continuous on $(0,\infty)$, and for every $N\ge0$ there is a constant $C'_N$ such that
  $\abs{\theta_{\Phi,\omega}(r)-\delta(\omega)\Phi(0)}\le C'_N\,r^{-N}$ for all $r\ge1$.
\end{enumerate}

Identify $\Mat_2(R)$ with $R^4$ by the entries $(a,b,c,d)$ of $\begin{psmallmatrix}a&b\\c&d\end{psmallmatrix}$.
\begin{enumerate}
\item For a non-archimedean local field $F$, $\dd X$ is the additive Haar measure on $\Mat_2(F)$ with
  $\dd X(\Mat_2(\Ocal_F))=1$, and $\abs{\det g}_F^{-2}\dd X$ is the measure on $\GL_2(F)$ given by
  \[
    E\longmapsto\int_E\abs{\det g}_F^{-2}\,\dd X(g)\qquad(E\subseteq\GL_2(F)\text{ Borel}),
  \]
  where $\GL_2(F)$ is viewed as the open subset $\{\det\neq0\}$ of $\Mat_2(F)$
  (\cref{X.topgrp.thm:gl2-units-open-embedding}).
\item For $F=\R$, $\dd X$ is the Lebesgue measure of $\Mat_2(\R)\cong\R^4$.
\end{enumerate}

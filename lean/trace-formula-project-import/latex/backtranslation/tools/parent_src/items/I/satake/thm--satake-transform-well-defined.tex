Let $F$ be a non-archimedean local field with residue cardinality $q$, $G=\GL_2(F)$, $K=K_F$, and let $\dd x$ be the
normalized additive Haar measure on $F$.
\begin{enumerate}[label=(\alph*)]
\item The Satake transform $\Sat\colon\mathcal H(G,K)\to\C[X^{\pm1},Y^{\pm1}]$ is $\C$-linear.
\item Let $f\in\mathcal H(G,K)$ and $a,b\in\Z$. The function $n\mapsto f(t_{a,b}\,n)$ on $N(F)$ is right invariant under
  $N(\Ocal_F)=\{n(x):x\in\Ocal_F\}$ and vanishes outside finitely many cosets $n\,N(\Ocal_F)$; the function
  $x\mapsto f(t_{a,b}\,n(x))$ is integrable on $F$; and the coefficient of $X^aY^b$ in $\Sat f$ is
  \[
    q^{-(a-b)/2}\sum_{nN(\Ocal_F)\in N(F)/N(\Ocal_F)}f(t_{a,b}\,n)=q^{-(a-b)/2}\int_Ff(t_{a,b}\,n(x))\,\dd x .
  \]
\end{enumerate}

Let $\Omega\subseteq\A$ be compact, $\omega=\{n(u):u\in\Omega\}$ and $t_0>0$.
\begin{enumerate}
  \item[(a)] For every left Haar measure $\mu$ on $G(\A)$, $\mu^1(\mathfrak S(\omega,t_0))<\infty$
  (\cref{I.haar.def:haar-norm-one}).
  \item[(b)] Let $\dd g$, $\lambda$, $\nu$ and $c$ be as in \cref{I.haar.thm:iwasawa-integration}, and
  $\Omega+\Zhat=\{u+(0,z):u\in\Omega,\ z\in\Zhat\}\subseteq\A=\R\times\Af$. Then
  \[
    (\dd g)^1\bigl(\mathfrak S(\omega,t_0)\bigr)=\frac{4c\,\kappa(\nu)^2\,\lambda(\Omega+\Zhat)}{t_0}.
  \]
  In particular, for $\omega=\omega_0$ of \cref{I.red.thm:siegel-covering},
  $(\dd g)^1(\mathfrak S(\omega_0,t_0))=4\pi\prod_p\mu_p(K_p)/t_0$.
\end{enumerate}

Let $F$ be a non-archimedean local field with residue cardinality $q$, $G=\GL_2(F)$ and $K=K_F$. For
$(\alpha,\beta)\in(\C^\times)^2$ and a Laurent polynomial $P=\sum_{a,b}p_{a,b}X^aY^b$ write
$P(\alpha,\beta)=\sum_{a,b}p_{a,b}\alpha^a\beta^b$, and put $\omega_{\alpha,\beta}(f)=(\Sat f)(\alpha,\beta)$ for
$f\in\mathcal H(G,K)$.
\begin{enumerate}[label=(\alph*)]
\item $\omega_{\alpha,\beta}\colon\mathcal H(G,K)\to\C$ is a homomorphism of $\C$-algebras, with
  $\omega_{\alpha,\beta}(T_\unif)=q^{1/2}(\alpha+\beta)$ and $\omega_{\alpha,\beta}(R_\unif)=\alpha\beta$.
\item Every homomorphism of $\C$-algebras $\omega\colon\mathcal H(G,K)\to\C$ equals $\omega_{\alpha,\beta}$ for some
  $(\alpha,\beta)\in(\C^\times)^2$.
\item For $(\alpha,\beta),(\alpha',\beta')\in(\C^\times)^2$: $\omega_{\alpha,\beta}=\omega_{\alpha',\beta'}$ if and only if
  $(\alpha',\beta')=(\alpha,\beta)$ or $(\alpha',\beta')=(\beta,\alpha)$.
\end{enumerate}

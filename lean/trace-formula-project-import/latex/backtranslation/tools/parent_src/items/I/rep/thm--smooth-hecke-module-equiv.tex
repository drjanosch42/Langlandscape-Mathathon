Let $G$ be a locally profinite group with its Borel $\sigma$-algebra and $\mu$ a left Haar measure on $G$.
\begin{enumerate}[label=(\alph*)]
\item For every non-degenerate $C_c^\infty(G)$-module $(V,\cdot)$ there is a unique smooth representation $\pi$ of $G$
  on $V$ such that $\pi(f)v=f\cdot v$ for all $f\in C_c^\infty(G)$ and $v\in V$.
\item Let $\pi_1,\pi_2$ be smooth representations of $G$ on $V_1,V_2$ and $T\colon V_1\to V_2$ a $\C$-linear map. Then
  $T\pi_1(g)=\pi_2(g)T$ for all $g\in G$ if and only if $T\pi_1(f)=\pi_2(f)T$ for all $f\in C_c^\infty(G)$.
\end{enumerate}

Assume H3 (\cref{I.spec.hyp:langlands-l2}) and H2a (\cref{I.eis.hyp:eis-continuation}) for every Dirichlet pair $\mu$, every $n\in\Z$ and every $N\ge1$. Let
$f\in\Hecke\cap\HeckeTwo$ (\cref{I.hecke.def:test-function,I.hecke.def:convolution-span}, for the standard measure $\dd g$ of \cref{I.haar.def:haar-adelic}), let
$(N,F)$ and the bases $B_\mu$ be as in \cref{I.spec.def:eisenstein-kernel}, and let $o$ be an orbit (\cref{II.kernel.def:cuspidal-datum}). Write $K_{o,f}$ for the spectral
kernel of \cref{II.kernel.def:spectral-kernel} formed with the $B_\mu$. Then:
\begin{enumerate}[label=(\alph*)]
\item $K_{o,f}$ is continuous on $G(\A)^1\times G(\A)^1$;
\item let $f^*(x)=\overline{f(x^{-1})}$; for every $(N',F')$ and bases $B'_\mu$ as in \cref{I.spec.def:eisenstein-kernel} for $f^*$,
  the kernel $K_{o,f^*}$ formed with the $B'_\mu$ satisfies $K_{o,f^*}(x,y)=\overline{K_{o,f}(y,x)}$ for all $x,y\in G(\A)^1$;
\item let $g\in G(\A)^1$ with $g_\infty\in\Kinf$, and $\rho(g)f\colon z\mapsto f(zg)$; for every $(N',F')$ and bases $B'_\mu$ as in \cref{I.spec.def:eisenstein-kernel} for
  $\rho(g)f$, the kernel $K_{o,\rho(g)f}$ formed with the $B'_\mu$ satisfies $K_{o,f}(x,yg)=K_{o,\rho(g)f}(x,y)$ for all $x,y\in G(\A)^1$.
\end{enumerate}

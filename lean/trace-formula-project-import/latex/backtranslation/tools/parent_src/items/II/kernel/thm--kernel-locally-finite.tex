Let $f\colon G(\A)^1\to\C$ be continuous with compact support.
\begin{enumerate}
  \item[(a)] For every compact $C\subseteq G(\A)^1$ the set of $\gamma\in G(\Q)$ with $f(x^{-1}\gamma y)\neq0$ for some $x,y\in C$ is finite.
  \item[(b)] $K_f$ is continuous on $G(\A)^1\times G(\A)^1$.
  \item[(c)] $K_f(\delta x,\delta'y)=K_f(x,y)$ for all $\delta,\delta'\in G(\Q)$ and $x,y\in G(\A)^1$.
  \item[(d)] For every compact $C\subseteq G(\A)^1$ there are $c\ge0$ and a compact $C'\subseteq G(\A)^1$ such that
  $\abs{K_f(x,y)}\le c$ for all $x\in C$, $y\in G(\A)^1$, and $K_f(x,y)\neq0$ with $x\in C$ implies $y\in G(\Q)C'$.
\end{enumerate}

Let $f\colon G(\A)^1\to\C$ be continuous with compact support, $K_f$ its kernel (\cref{II.kernel.def:kernel}) and $K_{\mathcal O}$ the class kernels
(\cref{II.kernel.def:class-kernel}).
\begin{enumerate}
  \item[(a)] For every compact $C\subseteq G(\A)^1$ there is a finite set of classes outside of which $K_{\mathcal O}$ vanishes on $C\times C$, and for every
  class $\mathcal O$ only finitely many $\gamma\in\mathcal O$ have $f(x^{-1}\gamma y)\neq0$ for some $x,y\in C$.
  \item[(b)] $K_f(x,y)=\sum_{\mathcal O}K_{\mathcal O}(x,y)$ for all $x,y\in G(\A)^1$, a sum with finitely many non-zero terms.
  \item[(c)] Every $K_{\mathcal O}$ is continuous on $G(\A)^1\times G(\A)^1$, and $K_{\mathcal O}(\delta x,\delta y)=K_{\mathcal O}(x,y)$ for $\delta\in G(\Q)$.
\end{enumerate}

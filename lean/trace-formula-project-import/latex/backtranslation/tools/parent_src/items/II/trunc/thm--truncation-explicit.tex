Let $T\in\aB$ with $\alpha(T)\ge0$, let $\varphi\colon G(\A)^1\to\C$ be left $G(\Q)$-invariant, and let $x\in G(\A)^1$. With $F^G$ as in
\cref{II.trunc.def:compact-part} and $\Lambda^T$ as in \cref{II.trunc.def:truncation}:
\begin{enumerate}
  \item[(a)] if $F^G(x,T)=1$, then $\Lambda^T\varphi(x)=\varphi(x)$;
  \item[(b)] if $F^G(x,T)=0$, then there is $\delta_0\in G(\Q)$ with $\alpha(\HB(\delta_0x))>\alpha(T)$, and
  $\Lambda^T\varphi(x)=\varphi(x)-\varphi_B(\delta_0x)$ for every such $\delta_0$.
\end{enumerate}

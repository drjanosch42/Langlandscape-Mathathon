Let $T\in\aB$, let $F^G(\cdot,T)\colon G(\A)^1\to\R$ be the characteristic function of the compact part (\cref{II.trunc.def:compact-part}),
and let $p\colon G(\A)^1\to G(\Q)\backslash G(\A)^1$ be the quotient map. Then:
\begin{enumerate}
  \item[(a)] $F^G(\cdot,T)$ is Borel;
  \item[(b)] $F^G(\gamma xk,T)=F^G(x,T)$ for all $\gamma\in G(\Q)$, $x\in G(\A)^1$ and $k\in K$;
  \item[(c)] the set $p\bigl(\{x\in G(\A)^1:F^G(x,T)=1\}\bigr)$ has compact closure in $G(\Q)\backslash G(\A)^1$;
  \item[(d)] if $T'\in\aB$ and $\alpha(T)\le\alpha(T')$, then $F^G(x,T)\le F^G(x,T')$ for all $x\in G(\A)^1$.
\end{enumerate}

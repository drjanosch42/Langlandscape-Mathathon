Let $a,b\in\Q^\times$ with $a\neq b$, $\gamma=\diag(a,b)$, and let $\mu$, $\nu_T$, $\nu$, $\lambda$, $\dd k$ and $c$ be as in
\cref{II.orb.thm:torus-quotient-integration}, so that $\nu$ is a measure on $G_\gamma(\A)^1\backslash G(\A)^1=T^1\backslash G(\A)^1$
(\cref{II.orb.thm:hyperbolic-centralizer}). Then:
\begin{enumerate}
  \item[(a)] $n(-u)\,\gamma\,n(u)=\gamma\,n\bigl((1-b/a)u\bigr)$ for every $u\in\A$, and $\abs{1-b/a}_\A=1$;
  \item[(b)] for every continuous $f\colon G(\A)^1\to\C$ with compact support, the function $(u,k)\mapsto f(k^{-1}\gamma n(u)k)$ on $\A\times K$ is continuous
  with compact support, $x\mapsto f(x^{-1}\gamma x)$ is $\nu$-integrable on $T^1\backslash G(\A)^1$, and
  \[
    O_\gamma(f;\nu)=c\int_K\int_\A f\bigl(k^{-1}\gamma\,n(u)\,k\bigr)\,\dd\lambda(u)\,\dd k
  \]
  (\cref{II.orb.def:orbital-integral}).
\end{enumerate}

Let $f$, $\dd u$ and $K_B$ be as in \cref{II.kernel.def:constant-term-kernel}.
\begin{enumerate}
  \item[(a)] For every compact $C\subseteq G(\A)^1$ there are a finite set $S\subseteq T(\Q)$ and a compact set $U\subseteq\A$ such that
  $f(x^{-1}\gamma n(u)y)\neq0$ with $x,y\in C$, $\gamma\in T(\Q)$, $u\in\A$ implies $\gamma\in S$ and $u\in U$. In particular, for all $x,y$ the integrand
  $u\mapsto\sum_{\gamma\in T(\Q)}f(x^{-1}\gamma n(u)y)$ is continuous with compact support.
  \item[(b)] $K_B$ is continuous on $G(\A)^1\times G(\A)^1$.
  \item[(c)] $K_B(\eta x,\eta'y)=K_B(x,y)$ for all $\eta,\eta'\in N(\A)T(\Q)$ and $x,y\in G(\A)^1$.
\end{enumerate}

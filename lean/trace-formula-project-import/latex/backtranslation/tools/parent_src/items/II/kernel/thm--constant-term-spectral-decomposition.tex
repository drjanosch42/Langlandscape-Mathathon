Let $f\in\Hecke$ (\cref{I.hecke.def:test-function}), $N\ge1$, and assume $f(y\,(1,k))=f(y)$ for all $y\in G(\A)^1$ and $k\in\Kf(N)$
(\cref{I.topgrp.def:congruence-subgroups}). Let $\dd u$ be a Haar measure on $\A$, $K_B$ the constant-term kernel of $f$ formed with $\dd u$
(\cref{II.kernel.def:constant-term-kernel}), $K_{B,o,f}$ as in \cref{II.kernel.def:datum-constant-term-kernel} (with the same $\dd u$), and $S(N)$ the finite set of orbits
of \cref{II.kernel.thm:finite-spectral-support}. Then for all $x,y\in G(\A)^1$:
\begin{enumerate}[label=(\alph*)]
\item if $o$ is an orbit and some $\mu\in o$ has $\operatorname{cond}(\mu_1)\nmid N$ or $\operatorname{cond}(\mu_2)\nmid N$ (\cref{I.eis.def:conductor}), then
  $K_{B,o,f}(x,y)=0$;
\item $\displaystyle K_B(x,y)=\sum_{o\in S(N)}K_{B,o,f}(x,y)$.
\end{enumerate}

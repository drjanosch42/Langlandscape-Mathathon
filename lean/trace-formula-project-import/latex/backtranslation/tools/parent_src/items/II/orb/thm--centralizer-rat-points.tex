Let $\gamma\in\GL_2(\Q)$ and $\iota=\iota_\A\colon\GL_2(\Q)\to G(\A)$. Then:
\begin{enumerate}
  \item[(a)] for $\delta\in\GL_2(\Q)$, $\iota(\delta)\in G_\gamma(\A)$ if and only if $\delta\in G_\gamma(\Q)$;
  \item[(b)] $\iota(G_\gamma(\Q))\subseteq G_\gamma(\A)^1$;
  \item[(c)] $\iota(G_\gamma(\Q))$ is a discrete subgroup of $G_\gamma(\A)^1$.
\end{enumerate}

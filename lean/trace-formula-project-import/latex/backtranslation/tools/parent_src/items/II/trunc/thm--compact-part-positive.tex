Let $\mu$ be a left Haar measure on $G(\A)$, $\dd\dot x$ the measure on $X=G(\Q)\backslash G(\A)^1$ of
\texttt{I.quot.def:automorphic-quotient-measure}, $p\colon G(\A)^1\to X$ the quotient map, and $T\in\aB$. Put
\[
  X_1(T)=p\bigl(\{x\in G(\A)^1:F^G(x,T)=1\}\bigr),\qquad X_0(T)=p\bigl(\{x\in G(\A)^1:F^G(x,T)=0\}\bigr),
\]
with $F^G$ as in \cref{II.trunc.def:compact-part}. Then $X_1(T)$ is closed, $X_0(T)$ is open, $X$ is the disjoint union of $X_1(T)$ and $X_0(T)$, and:
\begin{enumerate}
  \item[(a)] $X_0(T)$ has positive $\dd\dot x$-measure;
  \item[(b)] if $\alpha(T)\ge0$, then $X_1(T)$ has positive $\dd\dot x$-measure.
\end{enumerate}

Let $\gamma\in\GL_2(\Q)$, let $\mu$ be a left Haar measure on $G(\A)$ and $\nu_\gamma$ a left Haar measure on $G_\gamma(\A)$, let $\nu$ be a Weil measure for
$(\mu,\nu_\gamma)$ on $G_\gamma(\A)\backslash G(\A)$, and let $\nu^1$ be a Weil measure for $(\mu^1,\nu_\gamma^1)$ on $G_\gamma(\A)^1\backslash G(\A)^1$
(\texttt{X.quot.def:weil-measure}; $\mu^1$ and $\nu_\gamma^1$ as in \texttt{I.haar.def:haar-norm-one} and \cref{II.orb.def:centralizer-haar-norm-one}). Then:
\begin{enumerate}
  \item[(a)] for every $\Phi\colon G(\A)\to\C$ with $\Phi(hx)=\Phi(x)$ for all $h\in G_\gamma(\A)$, $x\in G(\A)$, the function $\Phi$ is $\nu$-integrable on
  $G_\gamma(\A)\backslash G(\A)$ if and only if its restriction to $G(\A)^1$ is $\nu^1$-integrable on $G_\gamma(\A)^1\backslash G(\A)^1$, and
  \[
    \int_{G_\gamma(\A)\backslash G(\A)}\Phi\,\dd\nu=\int_{G_\gamma(\A)^1\backslash G(\A)^1}\Phi\,\dd\nu^1 ;
  \]
  \item[(b)] in particular, for every $F\colon G(\A)\to\C$,
  $\displaystyle\int_{G_\gamma(\A)\backslash G(\A)}F(x^{-1}\gamma x)\,\dd\nu(x)=O_\gamma\bigl(F|_{G(\A)^1};\nu^1\bigr)$ (\cref{II.orb.def:orbital-integral}).
\end{enumerate}

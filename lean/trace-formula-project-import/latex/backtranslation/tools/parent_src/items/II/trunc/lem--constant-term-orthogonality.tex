Let $\mu$ be a left Haar measure on $G(\A)$ and $\dd y$ the quotient measure of $\mu^1$ on $B(\Q)\backslash G(\A)^1$ (\texttt{X.quot.def:quotient-measure}). Let
$w\colon\R\to[0,\infty)$ be Borel and put $W(y)=w(\alpha(\HB(y)))$ for $y\in G(\A)^1$. Let $\varphi,\psi\colon G(\A)^1\to\C$ be Borel and left $B(\Q)$-invariant, with
$\int_{B(\Q)\backslash G(\A)^1}\abs\varphi^2W\,\dd y<\infty$ and $\int_{B(\Q)\backslash G(\A)^1}\abs\psi^2W\,\dd y<\infty$, and let $\varphi_B,\psi_B$ be their constant terms
(\texttt{I.cusp.def:constant-term}). Then:
\begin{enumerate}
  \item[(a)] $\displaystyle\int_{B(\Q)\backslash G(\A)^1}\abs{\varphi-\varphi_B}^2W\,\dd y\le\int_{B(\Q)\backslash G(\A)^1}\abs\varphi^2W\,\dd y$;
  \item[(b)] the functions $(\varphi-\varphi_B)\,\bar\psi\,W$ and $\varphi\,\overline{(\psi-\psi_B)}\,W$ are integrable on $B(\Q)\backslash G(\A)^1$, and
  \[
    \int_{B(\Q)\backslash G(\A)^1}(\varphi-\varphi_B)\,\bar\psi\,W\,\dd y=\int_{B(\Q)\backslash G(\A)^1}\varphi\,\overline{(\psi-\psi_B)}\,W\,\dd y .
  \]
\end{enumerate}

Let $T\in\aB$, let $\varphi\colon G(\A)^1\to\C$ satisfy $\varphi(\gamma x)=\varphi(x)$ for all $\gamma\in G(\Q)$ and $x\in G(\A)^1$, and let
$\Lambda^T\varphi$ be its truncation (\cref{II.trunc.def:truncation}), with constant terms $\varphi_B$ as in \texttt{I.cusp.def:constant-term}.
Then:
\begin{enumerate}
  \item[(a)] for $\delta\in G(\Q)$, $b\in B(\Q)$ and $x\in G(\A)^1$,
  $\varphi_B(b\delta x)\,\hat\tau_B(\HB(b\delta x)-T)=\varphi_B(\delta x)\,\hat\tau_B(\HB(\delta x)-T)$; so the summand of $\Lambda^T\varphi(x)$
  indexed by a coset $B(\Q)\delta$ does not depend on the representative $\delta$;
  \item[(b)] for $x\in G(\A)^1$ only finitely many cosets $B(\Q)\delta$ have $\hat\tau_B(\HB(\delta x)-T)\neq0$, and if $\delta_1,\dots,\delta_r\in G(\Q)$
  lie in pairwise distinct cosets and every such coset is among the $B(\Q)\delta_i$, then
  \[
    \Lambda^T\varphi(x)=\varphi(x)-\sum_{i=1}^r\varphi_B(\delta_ix)\,\hat\tau_B(\HB(\delta_ix)-T);
  \]
  \item[(c)] $\Lambda^T\varphi(\gamma x)=\Lambda^T\varphi(x)$ for all $\gamma\in G(\Q)$ and $x\in G(\A)^1$;
  \item[(d)] for $k\in K$ the function $\varphi_k(x)=\varphi(xk)$ satisfies $\Lambda^T(\varphi_k)(x)=(\Lambda^T\varphi)(xk)$ for all $x\in G(\A)^1$;
  \item[(e)] $\Lambda^T(a\varphi)=a\,\Lambda^T\varphi$ for $a\in\C$; and if $\psi\colon G(\A)^1\to\C$ is also left $G(\Q)$-invariant and, for every
  $x\in G(\A)^1$, the functions $u\mapsto\varphi(n(u)x)$ and $u\mapsto\psi(n(u)x)$ are integrable on $D=[0,1)\times\Zhat$, then
  $\Lambda^T(\varphi+\psi)=\Lambda^T\varphi+\Lambda^T\psi$.
\end{enumerate}

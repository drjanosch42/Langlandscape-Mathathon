Let $p$ be a prime, $\dd g_p$ the Haar measure on $\GL_2(\Qp)$ with $\vol(K_p)=1$ (\texttt{I.haar.def:haar-local}) and $\mathbf 1_{K_p}$ the indicator function of $K_p$.
\begin{enumerate}
  \item[(a)] Let $\gamma$ be as in \cref{II.orb.lem:unramified-orbit-support}, $\nu_\gamma$ a Haar measure on $G_\gamma(\Qp)$ with $\nu_\gamma(G_\gamma(\Qp)\cap K_p)=1$, and
  $\nu$ a Weil measure for $(\dd g_p,\nu_\gamma)$. Then $O_{\gamma,p}(\mathbf 1_{K_p};\nu)=1$.
  \item[(b)] Let $\gamma=\diag(a,b)$ with $a\neq b\in\Q^\times$, $\abs a_p=\abs b_p=1$, $\nu_\gamma$ a Haar measure on $G_\gamma(\Qp)=T(\Qp)$ with $\nu_\gamma(T(\Zp))=1$, and $\nu$
  a Weil measure for $(\dd g_p,\nu_\gamma)$. Then $O_{\gamma,p}(\mathbf 1_{K_p};\nu)=\abs{D(\gamma)}_p^{-1/2}$, $D(\gamma)=(a-b)^2/(ab)$.
\end{enumerate}
(\cref{II.orb.def:local-orbital-integral}.)

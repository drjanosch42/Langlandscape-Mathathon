Let $T\in\aB$ and $\Lambda^T$ as in \cref{II.trunc.def:truncation}.
\begin{enumerate}
  \item[(a)] If $\varphi\colon G(\A)^1\to\C$ is Borel, left $G(\Q)$-invariant and bounded on every compact subset of $G(\A)^1$, then so is
  $\Lambda^T\varphi$.
  \item[(b)] If $\varphi\colon G(\A)^1\to\C$ is continuous and left $G(\Q)$-invariant, then its constant term $\varphi_B$ (\texttt{I.cusp.def:constant-term})
  is continuous, and $\Lambda^T\varphi$ is Borel, left $G(\Q)$-invariant and bounded on every compact subset of $G(\A)^1$.
\end{enumerate}

Let $\mu$ be a left Haar measure on $G(\A)$, $X=G(\Q)\backslash G(\A)^1$ with the measure $\dd\dot x$ of \texttt{I.quot.def:automorphic-quotient-measure}, $p\colon G(\A)^1\to X$
the quotient map, and $T\in\aB$ with $\alpha(T)\ge0$. Let $P^T\colon\Ltwo(X)\to\Ltwo(X)$ be the map of \cref{II.trunc.def:truncation-l2}. Then:
\begin{enumerate}
  \item[(a)] for every $\varphi\in\Ltwo(X)$ (\texttt{I.quot.def:l2-automorphic}) and every Borel function $f\colon X\to\C$ in the class $\varphi$, the function
  $\Lambda^T(f\circ p)$ (\cref{II.trunc.def:truncation}) is left $G(\Q)$-invariant, the function it induces on $X$ is Borel and square-integrable, and its class in
  $\Ltwo(X)$ does not depend on the choice of $f$; it is $P^T\varphi$;
  \item[(b)] there is a continuous $\C$-linear operator $P\colon\Ltwo(X)\to\Ltwo(X)$ with $P\varphi=P^T\varphi$ for all $\varphi$, and $P$ is an orthogonal projection:
  $P\circ P=P$ and $P^*=P$;
  \item[(c)] $P^T\varphi=\varphi$ for every $\varphi\in\Lcusp$ (\texttt{I.cusp.def:l2-cusp}).
\end{enumerate}

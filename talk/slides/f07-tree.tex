\begin{frame}{Its Bruhat--Tits tree incarnation (stages 02--03)}
  \small
  \begin{center}
    \large $0\to H_1(\mathrm{BG})\to\ZZ[X_B(U')]^0\xrightarrow{\ \alpha\ }\big(\ZZ[X_B(U)]^0\big)^2\to0$\\[2pt]
    \normalsize the Ribet sequence, combinatorially \stattag{proved given connectivity}
  \end{center}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \textbf{Tree} (Ihara, Serre). Here $B=B_D$, and $q\nmid D$ is a \emph{second} prime ($D$, $q$ play Ribet's $q$, $p$):
      $\Gamma_U=\mathcal O_U[1/q]^1$ acts on the tree $\mathcal T_q$ of $\GL_2(\QQ_q)$;
      the quotient is the finite Brandt graph $\mathrm{BG}(U,q)$ (vertices $X_B(U)^{\sqcup2}$, edges $X_B(U\cap U_0(q))$), given connectivity.
    \item \textbf{SA} becomes a finite check: $\mathrm{BG}(U,q)$ connected\\
      (SA itself \stattag{cited}; in Lean a registered hypothesis).
    \item \textbf{Ihara}: $\ker\alpha^\vee=$ constants, over every abelian group \stattag{proved given connectivity}.
  \end{itemize}
  {\footnotesize Replace bad reduction by a group acting on a tree. The two-vertex graph $\GDR$ (edges $X_{B_q}(U)$, weights $w_e$; $X(\GDR)=\ZZ[E]^0$
  \stattag{proved}) is the combinatorial dual graph (identification at $\Gamma_0$-level: Deligne--Rapoport, Deuring, Ribet; along the tower: the comparison hypothesis Hyp.X, open).}

\begin{frame}{Setup}
  \small
  \begin{itemize}\setlength{\itemsep}{3pt}
    \item $B=B_D$: the \emph{definite} quaternion algebra over $\QQ$ ramified exactly at $\infty$ and at the primes of $D$
      (squarefree, odd number of prime factors); mostly $D=q$ prime.
    \item Levels: Eichler orders of level $M$, $(M,D)=1$, i.e.\ $U=U_0(M)$;\\
      and the tower $U_1(Mp^s)$, $s\ge0$.\\
      Forms: functions on the \emph{finite} class set $X_B(U)=B^\times\backslash\widehat B^\times/U$.
    \item The prime $p$: $p\ge5$, $p\nmid qM$; for the defect formula (H0):\\ $N=qM$, $p\nmid\varphi(N)$, so $q\not\equiv1\bmod p$.
    \item Weights $k\ge2$, trivial character; ordinary parts via $e=\lim U_p^{n!}$; $\Lambda=\mathcal O[[1+p\ZZ_p]]$.
    \item $\m$ non-Eisenstein; (H2): $\bar\rho_\m$ absolutely irreducible on $G_{\QQ(\sqrt{p^*})}$ and $p$-distinguished.
  \end{itemize}
\end{frame}

\begin{frame}{Stage 04--05: up the $p$-power tower, and the one geometric hypothesis}
  \small
  \textbf{Ordinary passage} (stage 04), along $U_1(Mp^s)$, $s\ge1$:
  \begin{itemize}\setlength{\itemsep}{2pt}
    \item $e$ kills the Eisenstein cokernels: $\deg\circ U_p=p\deg$ \stattag{proved}. E.g.\ $q=11$, $p=5$:
      $\Phicomb$ cyclic of order $10$, $250$ ($s=1,2$) with $U_p^*=5$, so $e=0$ on $\Phicomb\otimes\ZZ_5$ \stattag{computed}.
    \item $\mathbf M^{\mathrm{ord}}(U)$ is $\Lambda$-free, with control at every finite level \stattag{proved}.
      $\Lambda$-adic Ribet sequence \stattag{proved given connectivity of all $\mathrm{BG}(UU_s,q)$, from cited SA, HS}.
  \end{itemize}
  \textbf{The comparison hypothesis} $\mathrm{Hyp.X}(M;q)$ \stattag{stage 05; a registered hypothesis; open}:
  isomorphisms $\varphi_s:\mathcal X_q(J_s)\xrightarrow{\sim}\ZZ[S_s]^0$, $J_s=\mathrm{Jac}\,X(\Gamma_1(Mp^s)\cap\Gamma_0(q))$, $S_s=X_{B_q}(U_1(Mp^s))$,
  that are (X1) Hecke-equivariant ($U_q\mapsto+\Pi_q$), (X2) isometric for the monodromy pairing, (X3) natural in $s$, (X4) Galois-compatible.
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item It isolates \emph{all} geometry of the route; at $\Gamma_0$-level, weight $2$, its shadow is Ribet's theorem.
    \item Given its character-group part: $\varprojlim e(\mathcal X_q(J_s)\otimes\ZZ_p)\cong\mathbf M^{\mathrm{ord}}(M)$; pairings (c2) \stattag{open}.
  \end{itemize}
\end{frame}
